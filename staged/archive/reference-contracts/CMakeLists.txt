cmake_minimum_required(VERSION 3.20)
project(archive_reference_contracts LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "These contracts require the project Win32 MSVC native ABI")
endif()
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(variant legacy packed)
    set(target "archive_index_reference_${variant}")
    add_executable(${target} "../index_reference_tests.cpp")
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated" "${PROJECT_ROOT}/native")
    target_compile_options(${target} PRIVATE /W4 /fp:precise)
    if(variant STREQUAL "packed")
        target_compile_definitions(${target} PRIVATE COD4X1KHZ=1)
    endif()
endforeach()
